 424:     def _calculate_trend_slope(self, series: pd.Series) -> float:
 425:         """
 426:         Calculate linear trend slope for a time series window.
 427:         
 428:         Args:
 429:             series (pd.Series): Time series data window
 430:             
 431:         Returns:
 432:             float: Slope of linear trend
 433:         """
 434:         if len(series) < 2 or series.isnull().all():
 435:             return np.nan
 436:         
 437:         # Remove NaN values
 438:         clean_series = series.dropna()
 439:         if len(clean_series) < 2:
 440:             return np.nan
 441:         
 442:         # Create x values (time indices)
 443:         x = np.arange(len(clean_series))
 444:         y = clean_series.values
 445:         
 446:         # Calculate slope using least squares
 447:         try:
 448:             slope = np.polyfit(x, y, 1)[0]
 449:             return slope
 450:         except:
 451:             return np.nan

 453:     def create_feature_selection_methods(self, data: pd.DataFrame, 
 454:                                        target_column: str,
 455:                                        method: str = 'correlation',
 456:                                        top_k: Optional[int] = None) -> List[str]:
 457:         """
 458:         Create feature selection methods to identify most relevant features.
 459:         
 460:         Selects features based on correlation, mutual information, or variance.
 461:         
 462:         Args:
 463:             data (pd.DataFrame): Data with features
 464:             target_column (str): Target variable for feature selection
 465:             method (str): Selection method ('correlation', 'variance', 'mutual_info')
 466:         
 467:         Returns:
 468:             List[str]: List of selected feature names
 469:         """
 470:         if target_column not in data.columns:
 471:             self.logger.error(f"Target column '{target_column}' not found in data")
 472:             return []
 473: 
 474:         # Get feature columns (exclude target and non-numeric columns)
 475:         # Use a robust numeric check to include other numeric dtypes
 476:         feature_columns = [col for col in data.columns 
 477:                           if col != target_column and 
 478:                           is_numeric_dtype(data[col])]
 479: 
 480:         if not feature_columns:
 481:             self.logger.warning("No numeric feature columns found for selection")
 482:             return []
 483: 
 484:         selected_features = []
 485: 
 486:         if method == 'correlation':
 487:             # Select features based on correlation with target
 488:             correlations = data[feature_columns + [target_column]].corr()[target_column].abs()
 489:             correlations = correlations.drop(target_column).sort_values(ascending=False)
 490: 
 491:             # Select features with correlation > 0.1
 492:             selected_features = correlations[correlations > 0.1].index.tolist()
 493:             # Limit to top_k if requested
 494:             if top_k is not None:
 495:                 selected_features = selected_features[:top_k]
 496: 
 497:         elif method == 'variance':
 498:             # Select features with sufficient variance
 499:             variances = data[feature_columns].var()
 500:             # Select features with variance > 1% of maximum variance
 501:             threshold = variances.max() * 0.01
 502:             selected = variances.sort_values(ascending=False).index.tolist()
 503:             # If top_k provided, pick top_k by variance; otherwise use threshold rule
 504:             if top_k is not None:
 505:                 selected_features = selected[:top_k]
 506:             else:
 507:                 selected_features = variances[variances > threshold].index.tolist()
 508: 
 509:         elif method == 'mutual_info':
 510:             if not _SKLEARN_AVAILABLE:
 511:                 self.logger.warning("sklearn not available: falling back to correlation method for mutual_info")
 512:                 return self.create_feature_selection_methods(data, target_column, 'correlation')
 513: 
 514:             # Use mutual information regression to score features
 515:             X = data[feature_columns].fillna(method='ffill').fillna(method='bfill').fillna(0)
 516:             y = data[target_column].values
 517:             try:
 518:                 mi = mutual_info_regression(X, y, discrete_features=False)
 519:                 mi_series = pd.Series(mi, index=feature_columns).sort_values(ascending=False)
 520:                 # Select features with MI greater than the median
 521:                 selected_features = mi_series[mi_series > mi_series.median()].index.tolist()
 522:                 if top_k is not None:
 523:                     selected_features = selected_features[:top_k]
 524:             except Exception as e:
 525:                 self.logger.warning(f"Mutual information computation failed: {e}. Falling back to correlation.")
 526:                 return self.create_feature_selection_methods(data, target_column, 'correlation')
 527: 
 528:         elif method == 'correlation_matrix':
 529:             # Remove multicollinear features above a threshold (default 0.9)
 530:             selected_features = self.select_features_correlation_threshold(data[feature_columns], threshold=0.9)
 531:             if top_k is not None:
 532:                 selected_features = selected_features[:top_k]
 533: 
 534:         elif method == 'model_based':
 535:             if not _SKLEARN_AVAILABLE:
 536:                 self.logger.error("sklearn not available: cannot perform model-based selection")
 537:                 return feature_columns
 538:             # Use a simple random forest selector
 539:             X = data[feature_columns].fillna(method='ffill').fillna(method='bfill').fillna(0)
 540:             y = data[target_column].values
 541:             selected_features = self.select_features_model_based(X, y)
 542:             if top_k is not None:
 543:                 selected_features = selected_features[:top_k]
 544: 
 545:         else:
 546:             self.logger.error(f"Unknown feature selection method: {method}")

 552:     def select_features_correlation_threshold(self, data: pd.DataFrame, threshold: float = 0.9) -> List[str]:
 553:         """
 554:         Drop features that are highly correlated with each other above a threshold.
 555: 
 556:         Keeps the feature with the highest variance among correlated pairs.
 557:         """
 558:         if data.shape[1] == 0:
 559:             return []
 560: 
 561:         corr_matrix = data.corr().abs()
 562:         upper = corr_matrix.where(np.triu(np.ones(corr_matrix.shape), k=1).astype(bool))
 563:         to_drop = [column for column in upper.columns if any(upper[column] > threshold)]
 564: 
 565:         selected = [col for col in data.columns if col not in to_drop]
 566:         self.logger.info(f"Dropped {len(to_drop)} features with correlation above {threshold}")
 567:         return selected
