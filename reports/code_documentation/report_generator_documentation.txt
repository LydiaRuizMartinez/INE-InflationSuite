 181:         if isinstance(data.index, pd.DatetimeIndex):
 182:             # Ensure naive datetime index
 183:             hist_dates = data.index.tz_convert(None) if getattr(data.index, 'tz', None) is not None else data.index
 184:         elif 'fecha' in data.columns:
 185:             hist_dates = _to_naive_datetime(data['fecha'])
 186:         else:
 187:             hist_dates = pd.date_range(start='2010-01-01', periods=len(data), freq='M')

 197:             pred_dates = _to_naive_datetime(predictions['fecha'])

 682:     def _analyze_historical_data(self, data: pd.DataFrame) -> Dict[str, Any]:
 683:         """Analyze historical inflation data."""
 684:         inflation_col = self._find_inflation_column(data)
 685:         if not inflation_col:
 686:             return {'error': 'No inflation column found'}
 687:         
 688:         inflation_data = data[inflation_col].dropna()
 689:         
 690:         # Basic statistics
 691:         stats = {
 692:             'mean': float(np.mean(inflation_data)),
 693:             'median': float(np.median(inflation_data)),
 694:             'std': float(np.std(inflation_data)),
 695:             'min': float(np.min(inflation_data)),
 696:             'max': float(np.max(inflation_data)),
 697:             'count': len(inflation_data)
 698:         }
 699:         
 700:         # Periods analysis
 701:         periods = {
 702:             'high_inflation_periods': len(inflation_data[inflation_data > 3.0]),

 704:             'deflation_periods': len(inflation_data[inflation_data < 0]),
 705:             'stable_periods': len(inflation_data[(inflation_data >= 1.0) & (inflation_data <= 3.0)])
 706:         }
 707:         
 708:         # Volatility analysis
 709:         volatility = {
 710:             'coefficient_variation': stats['std'] / abs(stats['mean']) if stats['mean'] != 0 else 0,
 711:             'range': stats['max'] - stats['min'],
 712:             'iqr': float(np.percentile(inflation_data, 75) - np.percentile(inflation_data, 25))
 713:         }
 714:

 716:         if len(inflation_data) >= 12:
 717:             recent_12m = inflation_data.tail(12)

 721:                 'recent_12m_avg': float(np.mean(recent_12m)),

 723:                 'trend_direction': 'increasing' if np.mean(recent_12m) > np.mean(previous_12m) else 'decreasing',
 724:                 'trend_magnitude': float(abs(np.mean(recent_12m) - np.mean(previous_12m)))
 725:             }
 726:         else:

 730:             'statistics': stats,
