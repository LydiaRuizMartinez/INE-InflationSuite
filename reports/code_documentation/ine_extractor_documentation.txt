 101:     def _make_request_with_retry(self, url: str, params: Optional[Dict] = None) -> requests.Response:
 102:         """
 103:         Make HTTP request with exponential backoff retry logic.
 104:         
 105:         Args:
 106:             url: Target URL for the request
 107:             params: Optional query parameters
 108:             
 109:         Returns:
 110:             HTTP response object
 111:             
 112:         Raises:
 113:             requests.RequestException: If all retry attempts fail
 114:         """
 115:         last_exception = None
 116:         
 117:         for attempt in range(self.max_attempts):
 118:             try:
 119:                 self.logger.info(f"Attempting request to {url} (attempt {attempt + 1}/{self.max_attempts})")
 120:                 
 121:                 response = self.session.get(
 122:                     url, 
 123:                     params=params, 
 124:                     timeout=self.timeout
 125:                 )
 126:                 response.raise_for_status()
 127:                 
 128:                 self.logger.info(f"Request successful on attempt {attempt + 1}")
 129:                 return response
 130:                 
 131:             except requests.exceptions.RequestException as e:
 132:                 last_exception = e
 133:                 self.logger.warning(f"Request failed on attempt {attempt + 1}: {e}")
 134:                 
 135:                 if attempt < self.max_attempts - 1:
 136:                     wait_time = self.backoff_factor ** attempt
 137:                     self.logger.info(f"Waiting {wait_time} seconds before retry...")
 138:                     time.sleep(wait_time)
 139:                     
 140:         # If we get here, all attempts failed
 141:         self.logger.error(f"All {self.max_attempts} attempts failed for URL: {url}")
 142:         raise last_exception
