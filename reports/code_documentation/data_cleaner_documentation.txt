  76:         Uses linear interpolation for numeric columns and forward fill

  79:         Args:
  80:             data (pd.DataFrame): Input data with potential missing values
  81:             
  82:         Returns:
  83:             pd.DataFrame: Data with missing values handled
  84:         """
  85:         # Create a copy to avoid modifying original data
  86:         cleaned_data = data.copy()
  87:         
  88:         # Log initial missing values
  89:         initial_missing = cleaned_data.isnull().sum().sum()
  90:         if initial_missing > 0:
  91:             self.logger.info(f"Found {initial_missing} missing values to handle")
  92:         
  93:         # Handle numeric columns with linear interpolation
  94:         numeric_columns = cleaned_data.select_dtypes(include=[np.number]).columns
  95:         for col in numeric_columns:
  96:             missing_count = cleaned_data[col].isnull().sum()
  97:             if missing_count > 0:
  98:                 # Use linear interpolation for time series data
  99:                 cleaned_data[col] = cleaned_data[col].interpolate(
 100:                     method='linear',
 101:                     limit_direction='both'
 102:                 )
 103:                 self.logger.info(f"Interpolated {missing_count} missing values in column '{col}'")
 104:         
 105:         # Handle categorical columns with forward fill
 106:         categorical_columns = cleaned_data.select_dtypes(include=['object', 'category']).columns
 107:         for col in categorical_columns:
 108:             missing_count = cleaned_data[col].isnull().sum()
 109:             if missing_count > 0:
 110:                 cleaned_data[col] = cleaned_data[col].fillna(method='ffill')
 111:                 # If still missing values at the beginning, use backward fill
 112:                 cleaned_data[col] = cleaned_data[col].fillna(method='bfill')
 113:                 self.logger.info(f"Forward/backward filled {missing_count} missing values in column '{col}'")

 124:     def detect_outliers(self, data: pd.DataFrame) -> List[int]:
 125:         """
 126:         Detect outliers using IQR and Z-score methods.
 127:         
 128:         Combines both IQR (Interquartile Range) and Z-score methods to identify
 129:         outliers in numeric columns. Returns indices of rows containing outliers.
 130:         
 131:         Args:
 132:             data (pd.DataFrame): Input data to analyze for outliers
 133:             
 134:         Returns:
 135:             List[int]: List of row indices that contain outliers
 136:         """
 137:         outlier_indices = set()
 138:         numeric_columns = data.select_dtypes(include=[np.number]).columns
 139:         
 140:         for col in numeric_columns:
 141:             # Skip columns with all NaN values
 142:             if data[col].isnull().all():
 143:                 continue
 144:                 
 145:             # IQR method
 146:             Q1 = data[col].quantile(0.25)
 147:             Q3 = data[col].quantile(0.75)
 148:             IQR = Q3 - Q1
 149:             
 150:             # Define outlier bounds (1.5 * IQR is standard)
 151:             lower_bound = Q1 - 1.5 * IQR
 152:             upper_bound = Q3 + 1.5 * IQR
 153:             
 154:             # Find IQR outliers
 155:             iqr_outliers = data[(data[col] < lower_bound) | (data[col] > upper_bound)].index
 156:             outlier_indices.update(iqr_outliers)
 157:             
 158:             # Z-score method (threshold of 3 is standard)
 159:             z_scores = np.abs((data[col] - data[col].mean()) / data[col].std())
 160:             z_outliers = data[z_scores > 3].index
 161:             outlier_indices.update(z_outliers)
 162:             
 163:             # Log findings for this column
 164:             iqr_count = len(iqr_outliers)
 165:             z_count = len(z_outliers)
 166:             if iqr_count > 0 or z_count > 0:
 167:                 self.logger.info(f"Column '{col}': {iqr_count} IQR outliers, {z_count} Z-score outliers")
 168:         
 169:         outlier_list = sorted(list(outlier_indices))
 170:         self.logger.info(f"Total unique outlier rows detected: {len(outlier_list)}")
 171:         
 172:         return outlier_list
