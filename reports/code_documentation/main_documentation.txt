 227:         """Background thread for continuous resource monitoring."""
 228:         while self.monitoring_active:
 229:             try:
 230:                 process = psutil.Process()
 231:                 
 232:                 # System resources

 236:                 # Process resources
 237:                 memory_info = process.memory_info()
 238:                 process_memory_mb = memory_info.rss / 1024 / 1024
 239:                 process_cpu = process.cpu_percent()
 240:                 
 241:                 # Store resource snapshot
 242:                 resource_snapshot = {
 243:                     'timestamp': time.time(),
 244:                     'system_memory_percent': system_memory.percent,

 247:                     'process_memory_mb': process_memory_mb,
 248:                     'process_cpu_percent': process_cpu
 249:                 }
 250:                 
 251:                 self.resource_history.append(resource_snapshot)
 252:                 
 253:                 # Keep only last 100 snapshots to prevent memory buildup
 254:                 if len(self.resource_history) > 100:
 255:                     self.resource_history = self.resource_history[-100:]
 256:                 
 257:                 # Check for resource warnings
 258:                 self._check_resource_warnings(resource_snapshot)
 259:

 280:         """Optimize memory usage after each step."""

 293:             'memory_before_mb': memory_before,
 294:             'memory_after_mb': memory_after,

 300:

 794:         base_estimates = {
 795:             'data_extraction': 60,
