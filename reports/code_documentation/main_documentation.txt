 280:                 if len(self.resource_history) > 100:

 305:     def _optimize_memory_usage(self, step_name: str) -> Dict[str, Any]:
 306:         """Optimize memory usage after each step."""
 307:         # Get memory before optimization
 308:         process = psutil.Process()
 309:         memory_before = process.memory_info().rss / 1024 / 1024
 310:         
 311:         # Force garbage collection
 312:         collected = gc.collect()
 313:         
 314:         # Get memory after optimization
 315:         memory_after = process.memory_info().rss / 1024 / 1024
 316:         memory_freed = memory_before - memory_after
 317:         
 318:         optimization_result = {
 319:             'memory_before_mb': memory_before,
 320:             'memory_after_mb': memory_after,
 321:             'memory_freed_mb': memory_freed,
 322:             'objects_collected': collected
 323:         }
 324:         
 325:         self.performance_metrics['memory_optimization'][step_name] = optimization_result
 326:         
 327:         if memory_freed > 10:  # Only log if significant memory was freed
 328:             self.logger.info(f"Memory optimization for '{step_name}': "
 329:                            f"freed {memory_freed:.1f}MB, collected {collected} objects")

 818:         """Estimate total pipeline duration based on system resources."""
 819:         # Base estimates in seconds for each step
 820:         base_estimates = {
 821:             'data_extraction': 60,
 822:             'data_processing': 30,
 823:             'feature_engineering': 45,

 829:         # Adjust based on system resources
 830:         system_memory = psutil.virtual_memory()

 833:         # Memory adjustment factor (less memory = slower)

 835:         
 836:         # CPU adjustment factor (fewer cores = slower)
 837:         cpu_factor = max(0.5, min(2.0, cpu_count / 4))  # 4 cores baseline
 838:         
 839:         # Calculate adjusted estimates
 840:         total_estimate = 0

 842:             adjusted_time = base_time / (memory_factor * cpu_factor)
