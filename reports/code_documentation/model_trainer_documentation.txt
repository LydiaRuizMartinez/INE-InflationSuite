 205:     def _make_stationary(self, ts_data: pd.Series) -> Tuple[pd.Series, int]:
 206:         """
 207:         Make time series stationary using differencing.
 208:         
 209:         Args:
 210:             ts_data (pd.Series): Time series data
 211:             
 212:         Returns:
 213:             Tuple[pd.Series, int]: Stationary series and differencing order
 214:         """
 215:         max_d = self.arima_params.get('max_d', 2)
 216:         diff_order = 0
 217:         current_series = ts_data.copy()
 218:         
 219:         # Test stationarity
 220:         for d in range(max_d + 1):
 221:             adf_result = adfuller(current_series.dropna())
 222:             p_value = adf_result[1]
 223:             
 224:             if p_value <= 0.05:  # Series is stationary
 225:                 self.logger.info(f"Series is stationary with differencing order: {diff_order}")
 226:                 break
 227:             
 228:             if d < max_d:
 229:                 current_series = current_series.diff().dropna()
 230:                 diff_order += 1
 231:         
 232:         return current_series, diff_order

 471:     def _prepare_ml_data(self, X: pd.DataFrame, y: pd.Series) -> Tuple[pd.DataFrame, pd.Series]:
 472:         """
 473:         Prepare data for machine learning models.
 474:         
 475:         Args:
 476:             X (pd.DataFrame): Feature matrix
 477:             y (pd.Series): Target variable
 478:             
 479:         Returns:
 480:             Tuple[pd.DataFrame, pd.Series]: Cleaned feature matrix and target
 481:         """
 482:         # Align indices
 483:         common_idx = X.index.intersection(y.index)
 484:         X_aligned = X.loc[common_idx].copy()
 485:         y_aligned = y.loc[common_idx].copy()
 486: 
 487:         # Ensure numeric dtype and sanitize infinities
 488:         # Coerce non-numeric to NaN (should already be numeric, but safe)
 489:         for col in X_aligned.columns:
 490:             if not np.issubdtype(X_aligned[col].dtype, np.number):
 491:                 X_aligned[col] = pd.to_numeric(X_aligned[col], errors='coerce')
 492:         # Replace +/-inf with NaN
 493:         X_aligned.replace([np.inf, -np.inf], np.nan, inplace=True)
 494:         try:
 495:             y_aligned = pd.to_numeric(y_aligned, errors='coerce')
 496:         except Exception:
 497:             pass
 498:         y_aligned.replace([np.inf, -np.inf], np.nan, inplace=True)
 499: 
 500:         # Handle missing values
 501:         X_clean = X_aligned.fillna(method='ffill').fillna(method='bfill')
 502:         # Any remaining NaN set to 0 for features
 503:         X_clean = X_clean.fillna(0)
 504:         y_clean = y_aligned.fillna(method='ffill').fillna(method='bfill')
 505: 
 506:         # Remove any remaining NaN or non-finite rows
 507:         finite_mask = np.isfinite(X_clean).all(axis=1) & np.isfinite(y_clean)
 508:         valid_mask = finite_mask & ~(X_clean.isnull().any(axis=1) | y_clean.isnull())
 509:         X_clean = X_clean[valid_mask]
 510:         y_clean = y_clean[valid_mask]
 511:         
 512:         self.logger.info(f"Prepared ML data: {len(X_clean)} samples, {len(X_clean.columns)} features")
 513:         
 514:         return X_clean, y_clean
